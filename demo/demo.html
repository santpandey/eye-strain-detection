/// Quality gates (initial values, tune with A5 eval).
pub const HEAD_POSE_DEG: f64 = 20.0;   // |yaw| or |pitch| → 'headPose'
pub const FACE_LUMA_MIN: f64 = 45.0;   // darker face → 'dark'
pub const PROM_START: f64 = 0.3;   // open a blink event when c − rest > this
pub const PROM_END: f64 = 0.08;    // close it when c − rest < this
pub const PEAK_COMPLETE: f64 = 0.8; // peak c for a complete blink
